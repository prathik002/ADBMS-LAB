2. Create a view for invoice products(name,inv_id,pid,quantity).
 mysql> CREATE VIEW v2 AS
    -> SELECT
    ->     c.name,
    ->     im.inv_id,
    ->     ii.pid,
    ->     ii.quantity
    -> FROM customer c
    -> JOIN invoice_master im ON c.id = im.cust_id
    -> JOIN invoice_item ii ON im.inv_id = ii.inv_id;
Query OK, 0 rows affected (0.01 sec)

mysql> select * from v2;

| name | inv_id | pid  | quantity |
