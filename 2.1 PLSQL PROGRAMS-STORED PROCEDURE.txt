2. Create a procedure  to find greatest of 3 numbers.
mysql> delimiter //
mysql> create procedure greatest_num(a int,b int,c int)
    -> begin
    -> declare max_num int;
    -> if a>=b and a>=c then
    -> set max_num=a;
    -> elseif b>=a and b>=c then
    -> set max_num=b;
    -> else
    -> set max_num=c;
    -> end if;
    -> select max_num as greatest_number;
    -> end //
Query OK, 0 rows affected (0.01 sec)

mysql> delimiter ;
mysql> call greatest_num(20,43,12);

    -> begin
    -> select * from invoice_master where cust_id=cid;
