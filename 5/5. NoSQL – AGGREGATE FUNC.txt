2.Write a MongoDB query to count number of customers in each city.
Invoice> db.Customer.aggregate([ { $group: { _id: "$city", count: { $sum: 1 } } } ])
[
  { _id: 'Calicut', count: 1 },
  { _id: 'Ernakulam', count: 1 },
  { _id: 'Thiruvananthapuram', count: 2 },
  { _id: 'Thrissur', count: 1 }
]



3.Write a MongoDB query to find minimum and maximum age in each city.
Invoice> db.Customer.aggregate([
|   {
|     $group: {
|       _id: "$city",
|       minAge: { $min: "$age" },
|       maxAge: { $max: "$age" }
|     }
|   },
|   {
|     $project: {
|       _id: 0,
|       city: "$_id",
|       minAge: 1,
|       maxAge: 1
|     }
|   }
| ])
[
  { minAge: 32, maxAge: 32, city: 'Thrissur' },
  { minAge: 35, maxAge: 35, city: 'Calicut' },
  { minAge: 45, maxAge: 45, city: 'Ernakulam' },
  { minAge: 29, maxAge: 55, city: 'Thiruvananthapuram' }
]
